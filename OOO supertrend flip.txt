//@version=5
strategy("3 Supertrend Flip", overlay=true, default_qty_type=strategy.percent_of_equity, default_qty_value=100, commission_type=strategy.commission.percent, commission_value=0.1)

atrLen = input.int(10, "ATR len", minval=1)
factor = input.float(3.0, "Factor", step=0.1)
tp = input.float(1.0, "TP ($)", step=0.1)
sl = input.float(0.5, "SL ($)", step=0.1)

[st, dir] = ta.supertrend(factor, atrLen)
plot(st, "Supertrend", color=dir < 0 ? color.green : color.red)

// dir < 0 が上昇トレンド
if ta.change(dir) < 0 and strategy.position_size == 0
    strategy.entry("Long", strategy.long)

if strategy.position_size > 0
    strategy.exit("Exit", "Long", limit=strategy.position_avg_price + tp, stop=strategy.position_avg_price - sl)
